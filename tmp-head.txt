const SR = 48000;

function rms(buf: Float32Array, from: number, to: number): number {
  const a = Math.floor(buf.length * from);
  const b = Math.floor(buf.length * to);
  let s = 0;
  let n = 0;
  for (let i = a; i < b; i++) {
    s += buf[i]! * buf[i]!;
    n++;
  }
  return n > 0 ? Math.sqrt(s / n) : 0;
}
